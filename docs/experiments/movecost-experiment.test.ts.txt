import { it } from 'vitest';
import { applyMoves, randomMoves, solvedState, toColorArray, formatMove, type Move } from '../src/core/cube';
import { solveCfop } from '../src/solve/cfop';
import { MOVE_COST, turnCost } from '../src/solve/cfop/moveCost';
import { solveF2LAdvanced } from '../src/solve/cfop/f2lAdvanced';
import { countRegrips } from '../src/solve/cfop/grip';
import { writeFileSync } from 'fs';

const solved = solvedState();
const CURRENT = { ...MOVE_COST };
const faces = 'R L U D F B r l u d f b M S E'.split(' ');
const all = faces.flatMap((b) => [b, `${b}'`, `${b}2`]);
const tbl = (o: Record<string, number>) => Object.fromEntries(all.map((m) => [m, o[m] ?? 2.6]));
const PROPOSED: Record<string, number> = {
  R: 1, "R'": 1, R2: 1.6, L: 1.1, "L'": 1.1, L2: 1.8,
  U: 1.1, "U'": 1.1, U2: 1.7, D: 1.5, "D'": 1.6, D2: 2.6,
  F: 1.4, "F'": 1.3, F2: 2.6, B: 2.2, "B'": 2.2, B2: 3.0,
  r: 1.1, "r'": 1.1, r2: 1.8, l: 1.3, "l'": 1.3, l2: 2.0,
  u: 1.4, "u'": 1.5, u2: 2.2, d: 2.0, "d'": 2.0, d2: 3.0,
  f: 1.7, "f'": 1.6, f2: 3.0, b: 2.8, "b'": 2.8, b2: 3.4,
  M: 1.6, "M'": 1.3, M2: 2.1, S: 1.6, "S'": 1.8, S2: 2.6, E: 1.6, "E'": 1.6, E2: 2.6,
};
// MCC defaults divided by the wrist turn (0.8): wide = face equivalent (my reading, not MCC).
const MCC: Record<string, number> = {
  R: 1, "R'": 1, R2: 1.65, L: 1, "L'": 1, L2: 1.65,
  U: 1.25, "U'": 1.25, U2: 2.06, D: 1.75, "D'": 1.75, D2: 2.89,
  F: 1.25, "F'": 1.56, F2: 2.89, B: 1.25, "B'": 1.25, B2: 2.89,
  M: 1.63, "M'": 1.25, M2: 2.06, S: 1.56, "S'": 1.95, S2: 2.58, E: 1.56, "E'": 1.95, E2: 2.58,
};
for (const w of 'rlufdb') for (const s of ['', "'", '2']) MCC[w + s] = MCC[w.toUpperCase() + s];
const UNIFORM = Object.fromEntries(all.map((m) => [m, 1]));
const TABLES: Record<string, Record<string, number>> = { current: tbl(CURRENT), proposed: tbl(PROPOSED), mcc: tbl(MCC), uniform: UNIFORM };

it('exp', () => {
  const name = process.env.TABLE ?? 'current';
  for (const k of Object.keys(MOVE_COST)) delete MOVE_COST[k];
  Object.assign(MOVE_COST, TABLES[name]);
  let seed = 12345;
  const rand = () => (seed = (seed * 16807) % 2147483647) / 2147483647;
  const N = 100;
  const tot: Record<string, number> = { moves: 0, regrips: 0, wide: 0, D: 0, F: 0, U2: 0, R2: 0 };
  const algs: string[] = [];
  const evalAlg: Move[][] = [];
  for (let n = 0; n < N; n++) {
    const state = applyMoves(solved, randomMoves(25, rand));
    const plan = solveCfop(state, { crossColors: ['W'], f2l: 'advanced', oll: 'full', pll: 'full' });
    let st = state;
    for (const step of plan.steps.filter((s) => s.stage !== 'f2l' && s.stage !== 'oll' && s.stage !== 'pll')) st = applyMoves(st, step.alg);
    const arr = toColorArray(st);
    const pairs = solveF2LAdvanced(arr, 'white', 'time');
    const alg = pairs.flatMap((p) => p.alg);
    evalAlg.push(alg);
    algs.push(alg.map(formatMove).join(' '));
    tot.moves += alg.length;
    tot.regrips += countRegrips(alg);
    for (const m of alg) { const s = formatMove(m);
      if ('rlufdbMSE'.includes(m.base)) tot.wide++; if (m.base === 'D') tot.D++; if (m.base === 'F') tot.F++; if (s === 'U2') tot.U2++; if (s === 'R2') tot.R2++; }
  }
  const scored: Record<string, number> = {};
  for (const [tn, t] of Object.entries(TABLES)) {
    for (const k of Object.keys(MOVE_COST)) delete MOVE_COST[k];
    Object.assign(MOVE_COST, t);
    scored[tn] = evalAlg.reduce((a, x) => a + turnCost(x), 0) / N;
  }
  const out = { table: name, avg: Object.fromEntries(Object.entries(tot).map(([k, v]) => [k, v / N])), timeUnder: scored, algs };
  writeFileSync(`${process.env.OUT}/${name}.json`, JSON.stringify(out));
  console.log(JSON.stringify({ ...out, algs: undefined }));
}, 600000);
